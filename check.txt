CREATE OR REPLACE MODEL `project_id.dataset_name.boosted_tree_model`
OPTIONS (
  model_type = 'BOOSTED_TREE_CLASSIFIER',
  input_label_cols = ['label'],

  -- Hyperparameters
  max_iterations = 100,         -- Number of boosting iterations
  learn_rate = 0.05,            -- Learning rate
  max_tree_depth = 6,           -- Maximum depth of each tree
  subsample = 0.7,              -- Row sampling ratio
  colsample_bytree = 0.8,       -- Column sampling ratio per tree
  min_split_loss = 1.0,         -- Minimum loss reduction required to make a further partition (gamma)
  l1_reg = 0.1,                 -- L1 regularization term
  l2_reg = 1.0,                 -- L2 regularization term
  data_split_method = 'AUTO',   -- Let BQML auto-split train/eval sets
  enable_global_explain = TRUE  -- Enables SHAP explainability
) AS
SELECT
  feature1,
  feature2,
  feature3,
  feature4,
  feature5,
  feature6,
  feature7,
  feature8,
  feature9,
  feature10,
  label
FROM
  `project_id.dataset_name.training_table`;

CREATE OR REPLACE TABLE `project_id.dataset_name.training_table` AS
SELECT
  CAST(RAND() * 10 AS FLOAT64) AS feature_float1,
  CAST(RAND() * 100 AS FLOAT64) AS feature_float2,
  CAST(RAND() * 50 AS FLOAT64) AS feature_float3,
  CAST(RAND() * 0.5 AS FLOAT64) AS feature_float4,
  CAST(RAND() * 200 AS FLOAT64) AS feature_float5,
  CAST(RAND() * 5 AS FLOAT64) AS feature_float6,
  CAST(RAND() * 300 AS FLOAT64) AS feature_float7,
  CAST(RAND() * 2 AS FLOAT64) AS feature_float8,
  CAST(RAND() * 3 AS FLOAT64) AS feature_float9,
  CAST(RAND() * 4 AS FLOAT64) AS feature_float10,

  CAST(FLOOR(RAND() * 10) AS INT64) AS feature_int1,
  CAST(FLOOR(RAND() * 100) AS INT64) AS feature_int2,
  CAST(FLOOR(RAND() * 3) AS INT64) AS feature_int3,
  CAST(FLOOR(RAND() * 7) AS INT64) AS feature_int4,
  CAST(FLOOR(RAND() * 15) AS INT64) AS feature_int5,

  -- Simple rule to generate binary label using a mix of features + noise
  IF(
    (feature_float1 + feature_float2 * 0.1 + feature_int1 * 0.05 + RAND()) > 10,
    1,
    0
  ) AS label
FROM
  UNNEST(GENERATE_ARRAY(1, 1000)) AS row_id;
