CREATE OR REPLACE TABLE `project_id.dataset_name.training_table` AS
WITH base AS (
  SELECT
    FORMAT('cust_%04d', row_num) AS customer_id,
    FORMAT_DATE('%Y-%m', DATE_SUB(DATE '2025-06-01', INTERVAL CAST(FLOOR(RAND() * 12) AS INT64) MONTH)) AS month_year,

    -- Float features
    RAND() * 10 AS feature_float1,
    RAND() * 100 AS feature_float2,
    RAND() * 50 AS feature_float3,
    RAND() * 0.5 AS feature_float4,
    RAND() * 200 AS feature_float5,
    RAND() * 5 AS feature_float6,
    RAND() * 300 AS feature_float7,
    RAND() * 2 AS feature_float8,
    RAND() * 3 AS feature_float9,
    RAND() * 4 AS feature_float10,

    -- Integer features
    CAST(FLOOR(RAND() * 10) AS INT64) AS feature_int1,
    CAST(FLOOR(RAND() * 100) AS INT64) AS feature_int2,
    CAST(FLOOR(RAND() * 3) AS INT64) AS feature_int3,
    CAST(FLOOR(RAND() * 7) AS INT64) AS feature_int4,
    CAST(FLOOR(RAND() * 15) AS INT64) AS feature_int5

  FROM UNNEST(GENERATE_ARRAY(1, 1000)) AS row_num
)

SELECT
  *,
  -- Adjusted threshold to make roughly half 1s and half 0s
  IF(
    feature_float1 + feature_float2 * 0.05 + feature_int1 * 0.1 + RAND() * 5 > 15,
    1,
    0
  ) AS label
FROM base;
-----------------------


SELECT
  predicted_label,
  predicted_label_probs,
  *
FROM
  ML.PREDICT(
    MODEL `project_id.dataset_name.boosted_tree_model`,
    (
      SELECT
        * EXCEPT(label)
      FROM
        `project_id.dataset_name.training_table`
      LIMIT 10
    )
  );
