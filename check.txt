CREATE OR REPLACE MODEL `project_id.dataset_name.boosted_tree_model`
OPTIONS (
  model_type = 'BOOSTED_TREE_CLASSIFIER',
  input_label_cols = ['label'],
  max_iterations = 100,
  learn_rate = 0.05,
  subsample = 0.8,
  colsample_bytree = 0.8,
  max_tree_depth = 6,
  l1_reg = 0.1,
  l2_reg = 1.0,
  enable_global_explain = TRUE
) AS
SELECT
  * EXCEPT(label)
FROM
  `project_id.dataset_name.training_table`;


======================

SELECT
  predicted_label,
  predicted_label_probs,
  *
FROM
  ML.PREDICT(
    MODEL `project_id.dataset_name.boosted_tree_model`,
    (
      SELECT
        * EXCEPT(label)
      FROM
        `project_id.dataset_name.training_table`
      LIMIT 10
    )
  );
